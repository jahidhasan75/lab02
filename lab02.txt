const students = [
  { name: "Jahid", marks: [85, 78, 92] },
  { name: "Rahim", marks: [65, 70, 58] },
  { name: "Karim", marks: [35, 40, 30] },
  { name: "Nadia", marks: [90, 95, 88] }
];

function calculateResult(student) {
  const total = student.marks.reduce((sum, mark) => sum + mark, 0);
  const average = total / student.marks.length;

  let grade;

  if (student.marks.some(mark => mark < 33)) {
    grade = "F";
  } else if (average >= 80) {
    grade = "A+";
  } else if (average >= 70) {
    grade = "A";
  } else if (average >= 60) {
    grade = "A-";
  } else if (average >= 50) {
    grade = "B";
  } else if (average >= 40) {
    grade = "C";
  } else {
    grade = "D";
  }

  return {
    name: student.name,
    total,
    average: average.toFixed(2),
    grade,
    result: grade === "F" ? "Fail" : "Pass"
  };
}

console.log("===== STUDENT RESULT SYSTEM =====");

students.forEach(student => {
  const result = calculateResult(student);

  console.log("------------------------------");
  console.log("Name:", result.name);
  console.log("Total Marks:", result.total);
  console.log("Average:", result.average);
  console.log("Grade:", result.grade);
  console.log("Result:", result.result);
});
